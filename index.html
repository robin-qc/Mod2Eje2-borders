/* ===== ESTILOS DE BORDER ===== */
border: 2px solid var(--neon-cyan);           /* Sólido */
border: 3px dashed var(--neon-green);        /* Discontinuo */
border: 3px dotted var(--neon-pink);         /* Punteado */
border: 5px double var(--neon-yellow);       /* Doble línea */
border: 5px groove var(--neon-purple);       /* 3D hundido */
border: 5px ridge var(--neon-orange);        /* 3D elevado */
border: 5px inset #00ffff;                  /* Contenido hundido */
border: 5px outset #ff00ff;                 /* Contenido elevado */
border: 3px hidden var(--neon-cyan);        /* Oculto */

/* ===== BORDES POR LADO ===== */
border-top: 5px solid var(--neon-cyan);
border-right: 3px dashed var(--neon-pink);
border-bottom: 4px dotted var(--neon-green);
border-left: 6px double var(--neon-yellow);

/* ===== BORDER-RADIUS ===== */
border-radius: 4px;                         /* Todos iguales */
border-radius: 4px 8px 12px 16px;           /* TL TR BR BL */
border-radius: 50%;                         /* Círculo perfecto */
border-radius: 50% / 30%;                   /* Elipse/Oval */

/* Individual corners */
border-top-left-radius: 10px;
border-top-right-radius: 20px;
border-bottom-right-radius: 30px;
border-bottom-left-radius: 40px;

/* ===== EFECTOS ESPECIALES ===== */
/* Gradiente border */
border: 3px solid transparent;
background: 
    linear-gradient(var(--bg-card), var(--bg-card)) padding-box,
    linear-gradient(45deg, var(--neon-cyan), var(--neon-pink)) border-box;

/* Glow effect */
border: 2px solid var(--neon-cyan);
box-shadow: 0 0 20px var(--neon-cyan-glow);

/* Animación de color */
border: 3px solid var(--neon-cyan);
animation: borderPulse 2s infinite;

@keyframes borderPulse {
    0%, 100% { border-color: var(--neon-cyan); }
    50% { border-color: var(--neon-pink); }
}

/* ===== TRUCOS AVANZADOS ===== */
/* Triángulo con borders */
.triangle {
    width: 0;
    height: 0;
    border-left: 50px solid transparent;
    border-right: 50px solid transparent;
    border-bottom: 100px solid var(--neon-cyan);
}

/* Border morphing */
.morph-border {
    border: 3px solid var(--neon-cyan);
    border-radius: 50% 50% 50% 50%;
    animation: morph 4s infinite;
}

@keyframes morph {
    0%, 100% { border-radius: 50% 50% 50% 50%; }
    25% { border-radius: 30% 70% 70% 30%; }
    50% { border-radius: 70% 30% 30% 70%; }
    75% { border-radius: 30% 70% 70% 30%; }
}
